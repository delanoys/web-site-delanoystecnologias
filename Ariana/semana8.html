2. Empaquetado y Desempaquetado

# Empaquetado
persona = ("Ariadna", 20, 1.65, True)

# Desempaquetado
nombre, edad, altura, es_estudiante = persona

print(f"Nombre: {nombre}")
print(f"Edad: {edad}")

3. Uso de Tuplas (Ventajas y Retornos Múltiples)

def calcular_estadisticas(notas):
    return (min(notas), max(notas), sum(notas)/len(notas))

notas = [4.2, 3.8, 4.5]
minimo, maximo, promedio = calcular_estadisticas(notas)
print(f"Min: {minimo}, Max: {maximo}, Prom: {promedio:.2f}")

Ejercicio 3

Crea una función que retorne una tupla con el área y perímetro de un rectángulo.

def rectangulo(base, altura):
    area = base * altura
    perimetro = 2 * (base + altura)
    return (area, perimetro)

a, p = rectangulo(8, 5)
print("Área:", a, "- Perímetro:", p)
