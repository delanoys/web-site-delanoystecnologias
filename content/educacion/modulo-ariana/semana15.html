Semana 15: Técnicas Iterativas y Búsqueda

1. Búsqueda Lineal

def busqueda_lineal(lista, objetivo):
    for i in range(len(lista)):
        if lista[i] == objetivo:
            return i
    return -1

numeros = [10, 25, 45, 67, 89, 32]
print(busqueda_lineal(numeros, 67))

2. Uso de 'in', index y métodos útiles

estudiantes = ["Ana", "Luis", "María", "Pedro"]

print("Luis" in estudiantes)
print(estudiantes.index("María"))

if "Carlos" not in estudiantes:
    print("Carlos no está en la lista")

3. Búsqueda en Diccionarios

notas = {"Ana": 4.5, "Luis": 3.8, "María": 4.2}

print(notas.get("Ana", "No encontrado"))
print(notas.get("Carlos", "No encontrado"))

Ejercicio 1

Implementa una función de búsqueda lineal que indique si un elemento existe en una lista.

def buscar_elemento(lista, objetivo):
    if objetivo in lista:
        return f"{objetivo} encontrado en la posición {lista.index(objetivo)}"
    return "Elemento no encontrado"

print(buscar_elemento([10, 20, 30, 40, 50], 30))

Ejercicio 3

Crea un programa que busque el número máximo y mínimo en una lista sin usar max() ni min().

def encontrar_max_min(lista):
    maximo = minimo = lista[0]
    for num in lista:
        if num > maximo:
            maximo = num
        if num < minimo:
            minimo = num
    return maximo, minimo

print(encontrar_max_min([45, 12, 89, 33, 67]))
